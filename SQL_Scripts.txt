CREATE TABLE client_details (
    id BIGINT NOT NULL AUTO_INCREMENT,
    client_identifier VARCHAR(20) NOT NULL UNIQUE,
    name VARCHAR(255),
    address VARCHAR(255),
    phone_number VARCHAR(255),
    create_date TIMESTAMP NOT NULL,
    last_update_date TIMESTAMP NULL,
    license_start_datetime TIMESTAMP NULL,
    license_end_datetime TIMESTAMP NULL,
    churn_rate_2_weeks DECIMAL(5,2),
    churn_rate_4_weeks DECIMAL(5,2),
    churn_rate_12_weeks DECIMAL(5,2),
    login_weekly_history VARCHAR(255),
    PRIMARY KEY (id)
);

    last_update_date TIMESTAMP NULL,
